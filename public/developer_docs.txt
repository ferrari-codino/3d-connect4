1. アプリケーション概要・システム仕様
--------------------------------------------------------------------------------
本アプリケーションは、4×4×4 の立体空間（計64マス・16本の支柱）で行う3D対戦型コネクトフォー（立体四目並べ）ゲームです。

■ 総合アクセスカウンター ＆ 統計分析ログエンジン
- リアルタイム総合アクセスカウンター（ヘッダー右上に配置）
- 統計データ活用・分析用ログ蓄積エンジン：
  * 日別集計（総訪問数、新規セッション数、モバイル/PC内訳、最終カウント、最終訪問時刻）
  * 時間帯別分布（0時〜23時）
  * デバイス別内訳（Mobile / Tablet / Desktop）
  * 将来の別画面表示・集計・エクスポート用API：
    - `window.getAccessStatistics()` : 統計サマリーオブジェクト取得
    - `window.exportAccessAnalyticsCSV()` : CSVスプレッドシート形式で一括ダウンロード
    - `window.exportAccessAnalyticsJSON()` : JSON形式でダウンロード
    - `window.printAccessStatistics()` : コンソール表形式出力

■ 基本ルール
- 4×4の16本の垂直な支柱に対して、プレイヤーとAI（またはプレイヤー同士）が交互に玉を投入します。
- 重力ルールにより、各支柱には下から順に最大4個まで玉が積み上がります。
- 縦・横・高さ・各種斜め（平面斜めおよび立体対角線）のいずれかで「自色の玉を4連続」で並べたプレイヤーが勝利となります。
- 計76通りのリーチ/勝利ライン（直線42本、平面対角線24本、立体対角線10本）が存在します。

■ アシスト機能（学習・戦術分析モード）
1. 【アシストモード１ (勝率予測・評価値)】:
   - 全16本の各支柱について、次に着手した場合の局面勝率予測（% / 必勝 / 必敗）をリアルタイム計算。
   - 3D盤面上空および左側支柱クイック選択ボタン上に色分けバッジ（緑: 有利、黄: 互角、赤: 不利）で表示。
2. 【アシストモード２ (戦術・リーチ・W/3連リーチ予告)】:
   - 立体四目並べの戦術パターンを検出・可視化:
     * 「王手（リーチ）」: 相手が次に置くと4連になるマスがある場合の防御警告、自分が4連完成になる勝利手。
     * 「ダブルリーチ狙い（両建て王手）」: 着手により次の手番で2方向の即時リーチを同時に作成する一手（必勝パターン）。
     * 「トリプルリーチ狙い（3連リーチ）」: 着手により次の手番で3方向以上の即時リーチを同時に作成する一手。
     * 「相手Wリーチ阻止（防ぎ所）」: 相手のダブルリーチの急所を先回りして塞ぐ好手。
     * 「危険手」: 上段に相手の即時勝利マスを献上してしまう自殺手。
   - 3Dキャンバス上および3Dステージ上部の「戦術アラートバナー」にて即時告知。

■ 待った機能（一手戻す）
- プレイヤーが直前の着手を取り消して指し直すことができる機能。
- 対局設定パネル、3Dステージツールバー、着手履歴ヘッダーの各所に配置。
- AIの思考中以外で着手履歴が存在するときに即座に1手巻き戻し。

■ デフォルト設定値
- 手番選択: 人間 先手（アクティブ表示）
- AIの強さ: 中級（アクティブ表示）
- コマのカラーテーマ: 黒 × 白（アクティブ表示）

    const WINNING_LINES = [];

    function initZobrist() {
      // Seeded random pseudo-number generator for deterministic Zobrist keys
      let seed = 0x12345678;
      for (let i = 0; i < 64 * 2; i++) {
        seed = (seed ^ (seed << 13)) >>> 0;
        seed = (seed ^ (seed >>> 17)) >>> 0;
        seed = (seed ^ (seed << 5)) >>> 0;
        ZOBRIST_TABLE[i] = seed;
      }
    }
    initZobrist();

    function computeZobristHash(b) {
      let h = 0;
      for (let i = 0; i < 64; i++) {
        const val = b[i];
        if (val === 1) h = (h ^ ZOBRIST_TABLE[i * 2]) >>> 0;
        else if (val === 2) h = (h ^ ZOBRIST_TABLE[i * 2 + 1]) >>> 0;
      }
      return h;
    }

      if (oImmediateThreats >= 2) return -8800; // 敵陣の2箇所同時王手（必敗）

      const currentKillers = (searchDepth < 20) ? killerMoves[searchDepth] : [-1, -1];

      const moveScores = validCols.map(col => {
        let score = 0;

        // 1. TT Best Move gets highest priority
        if (col === ttBestMove) {
          return { col, score: 20000 };
        }

        // 2. Killer moves priority
        if (col === currentKillers[0]) score += 8000;
        else if (col === currentKillers[1]) score += 6000;

        // 3. History heuristic priority
        score += historyTable[col];

        // 4. Quick positional & tactical static evaluation
        const z = h[col];
        const idx = col + z * 16;

        b[idx] = playerToMove;
        h[col]++;

        const win = checkWin(b);
        if (win && win.winner === playerToMove) {
          score += 15000;
        } else {
          score += evaluateBoard(b, h, playerToMove);
        }

        b[idx] = 0;
        h[col]--;

        return { col, score };
      });

      // Sort moves in descending order
      moveScores.sort((a, b) => b.score - a.score);
      return moveScores.map(m => m.col);
    }

    // Quiescence Search (静的静穏探索)
    function quiescenceSearch(b, h, alpha, beta, isMaximizing, evalAiPlayer, evalHumanPlayer, depthLimit = 2) {
      const standPat = evaluateBoard(b, h, evalAiPlayer);
      if (depthLimit <= 0) return standPat;

      if (isMaximizing) {
        if (standPat >= beta) return beta;
        alpha = Math.max(alpha, standPat);
      } else {
        if (standPat <= alpha) return alpha;
        beta = Math.min(beta, standPat);
      }

      const activePlayer = isMaximizing ? evalAiPlayer : evalHumanPlayer;
      const oppPlayer = isMaximizing ? evalHumanPlayer : evalAiPlayer;

      const validCols = [];
      for (let c = 0; c < 16; c++) {
        if (h[c] < 4) validCols.push(c);
      }

      // Find tactical threat moves (immediate 4-in-a-row wins or forced blocks)
      const tacticalCols = [];
      for (let col of validCols) {
        const z = h[col];
        const idx = col + z * 16;

        // Win check for active player
        b[idx] = activePlayer;
        const winAct = checkWin(b);
        b[idx] = 0;
        if (winAct) {
          return isMaximizing ? 9999 : -9999;
        }

        // Win check for opponent (block)
        b[idx] = oppPlayer;
        const winOpp = checkWin(b);
        b[idx] = 0;
        if (winOpp) {
          tacticalCols.push(col);
        }
      }

      if (tacticalCols.length === 0) {
        return standPat;
      }

      for (let col of tacticalCols) {
        const z = h[col];
        const idx = col + z * 16;
        b[idx] = activePlayer;
        h[col]++;

        const score = quiescenceSearch(b, h, alpha, beta, !isMaximizing, evalAiPlayer, evalHumanPlayer, depthLimit - 1);

        b[idx] = 0;
        h[col]--;

        if (isMaximizing) {
          if (score >= beta) return beta;
          alpha = Math.max(alpha, score);
        } else {
          if (score <= alpha) return alpha;
          beta = Math.min(beta, score);
        }
      }

      return isMaximizing ? alpha : beta;
    }

    function minimax(b, h, depth, alpha, beta, isMaximizing, evalAiPlayer, evalHumanPlayer, deadlineMs = null, currentPly = 0) {
      if (deadlineMs && Date.now() > deadlineMs) {
        return { score: evaluateBoard(b, h, evalAiPlayer), move: -1, aborted: true };
      }

      const winState = checkWin(b);
      if (winState) {
        if (winState.winner === evalAiPlayer) return { score: 9999 - currentPly, move: -1 };
        if (winState.winner === evalHumanPlayer) return { score: -9999 + currentPly, move: -1 };
      }

      if (depth <= 0) {
        // Apply Quiescence Search for quiet tactical stability
        const qScore = quiescenceSearch(b, h, alpha, beta, isMaximizing, evalAiPlayer, evalHumanPlayer, 2);
        return { score: qScore, move: -1 };
      }

      // Transposition Table Lookup
      const boardHash = computeZobristHash(b);
      const ttEntry = transpositionTable.get(boardHash);
      let ttBestMove = -1;

      if (ttEntry) {
        ttBestMove = ttEntry.move;
        if (ttEntry.depth >= depth) {
          if (ttEntry.flag === 'EXACT') {
            return { score: ttEntry.score, move: ttEntry.move };
          } else if (ttEntry.flag === 'LOWER' && ttEntry.score >= beta) {
            return { score: ttEntry.score, move: ttEntry.move };
          } else if (ttEntry.flag === 'UPPER' && ttEntry.score <= alpha) {
            return { score: ttEntry.score, move: ttEntry.move };
          }
        }
      }

      const activePlayer = isMaximizing ? evalAiPlayer : evalHumanPlayer;
      const oppPlayer = isMaximizing ? evalHumanPlayer : evalAiPlayer;

      const validCols = [];
      for (let c = 0; c < 16; c++) {
        if (h[c] < 4) validCols.push(c);
      }
      if (validCols.length === 0) {
        return { score: evaluateBoard(b, h, evalAiPlayer), move: -1 };
      }

      // 1. Immediate Win Check for active player
      for (let col of validCols) {
        const z = h[col];
        const idx = col + z * 16;
        b[idx] = activePlayer;
        const isWin = checkWin(b);
        b[idx] = 0;
        if (isWin) {
          return { score: isMaximizing ? (9999 - currentPly) : (-9999 + currentPly), move: col };
        }
      }

      // 2. Forced Block Check for opponent threats
      const forcedBlocks = [];
      for (let col of validCols) {
        const z = h[col];
        const idx = col + z * 16;
        b[idx] = oppPlayer;
        const isOppWin = checkWin(b);
        b[idx] = 0;
        if (isOppWin) {
          forcedBlocks.push(col);
        }
      }

      let moves = [];
      let isForced = false;
      if (forcedBlocks.length > 0) {
        if (forcedBlocks.length >= 2) {
          // Opponent has double threat -> Unblockable
          return { score: isMaximizing ? (-9999 + currentPly) : (9999 - currentPly), move: forcedBlocks[0] };
        }
        moves = forcedBlocks;
        isForced = true;
      } else {
        moves = getSortedValidMoves(b, h, activePlayer, ttBestMove, currentPly);
      }

      let bestMove = moves[0];
      const depthDec = isForced ? 0.5 : 1.0;
      const origAlpha = alpha;

      if (isMaximizing) {
        let maxEval = -Infinity;
        for (let i = 0; i < moves.length; i++) {
          if (deadlineMs && Date.now() > deadlineMs && i > 0) {
            return { score: maxEval, move: bestMove, aborted: true };
          }
          const col = moves[i];
          const z = h[col];
          const idx = col + z * 16;

          b[idx] = evalAiPlayer;
          h[col]++;

          const result = minimax(b, h, depth - depthDec, alpha, beta, false, evalAiPlayer, evalHumanPlayer, deadlineMs, currentPly + 1);

          b[idx] = 0;
          h[col]--;

          if (result.aborted && i > 0) {
            return { score: maxEval, move: bestMove, aborted: true };
          }

          if (result.score > maxEval) {
            maxEval = result.score;
            bestMove = col;
          }
          alpha = Math.max(alpha, result.score);
          if (beta <= alpha) {
            // Update Killer Moves & History Heuristic on Cutoff
            if (currentPly < 20 && col !== killerMoves[currentPly][0]) {
              killerMoves[currentPly][1] = killerMoves[currentPly][0];
              killerMoves[currentPly][0] = col;
            }
            historyTable[col] += depth * depth;
            break; // Alpha-Beta Cutoff
          }
        }

        // Store into Transposition Table
        let flag = 'EXACT';
        if (maxEval <= origAlpha) flag = 'UPPER';
        else if (maxEval >= beta) flag = 'LOWER';
        transpositionTable.set(boardHash, { depth, score: maxEval, flag, move: bestMove });

        return { score: maxEval, move: bestMove };
      } else {
        let minEval = Infinity;
        for (let i = 0; i < moves.length; i++) {
          if (deadlineMs && Date.now() > deadlineMs && i > 0) {
            return { score: minEval, move: bestMove, aborted: true };
          }
          const col = moves[i];
          const z = h[col];
          const idx = col + z * 16;

          b[idx] = evalHumanPlayer;
          h[col]++;

          const result = minimax(b, h, depth - depthDec, alpha, beta, true, evalAiPlayer, evalHumanPlayer, deadlineMs, currentPly + 1);

          b[idx] = 0;
          h[col]--;

          if (result.aborted && i > 0) {
            return { score: minEval, move: bestMove, aborted: true };
          }

          if (result.score < minEval) {
            minEval = result.score;
            bestMove = col;
          }
          beta = Math.min(beta, result.score);
          if (beta <= alpha) {
            // Update Killer Moves & History Heuristic on Cutoff
            if (currentPly < 20 && col !== killerMoves[currentPly][0]) {
              killerMoves[currentPly][1] = killerMoves[currentPly][0];
              killerMoves[currentPly][0] = col;
            }
            historyTable[col] += depth * depth;
            break; // Alpha-Beta Cutoff
          }
        }

        // Store into Transposition Table
        let flag = 'EXACT';
        if (minEval <= origAlpha) flag = 'UPPER';
        else if (minEval >= beta) flag = 'LOWER';
        transpositionTable.set(boardHash, { depth, score: minEval, flag, move: bestMove });

        return { score: minEval, move: bestMove };
      }
    }

    // Iterative Deepening Minimax Search Engine
    function getBestMoveIterative(b, h, targetMaxDepth, maxTimeMs, evalAiPlayer, evalHumanPlayer) {
      clearSearchCaches(); // Reset TT, Killer Moves, and History Table per search

      // 1. Opening Book Check
      const bookMove = getOpeningBookMove(b, h);
      if (bookMove !== -1) {
        return { move: bookMove, score: 500 };
      }

      const startTime = Date.now();
      const deadlineMs = startTime + maxTimeMs;

      const validCols = [];
      for (let c = 0; c < 16; c++) {
        if (h[c] < 4) validCols.push(c);
      }
      if (validCols.length === 0) return { move: -1, score: 0 };
      if (validCols.length === 1) return { move: validCols[0], score: 0 };

      // 2. Immediate Win Check for AI
      for (let col of validCols) {
        const z = h[col];
        const idx = col + z * 16;
        b[idx] = evalAiPlayer;
        const isWin = checkWin(b);
        b[idx] = 0;
        if (isWin) return { move: col, score: 9999 };
      }

      // 3. Prevent Opponent Win (Forced Block)
      const forcedBlocks = [];
      for (let col of validCols) {
        const z = h[col];
        const idx = col + z * 16;
        b[idx] = evalHumanPlayer;
        const isWin = checkWin(b);
        b[idx] = 0;
        if (isWin) forcedBlocks.push(col);
      }

      if (forcedBlocks.length === 1) {
        return { move: forcedBlocks[0], score: -500 };
      } else if (forcedBlocks.length >= 2) {
        return { move: forcedBlocks[0], score: -9999 };
      }

      // 4. Iterative Deepening Search
      let bestMove = validCols[0];
      let bestScore = -Infinity;

      for (let currentDepth = 2; currentDepth <= targetMaxDepth; currentDepth++) {
        if (Date.now() >= deadlineMs && currentDepth > 3) break;

        const result = minimax(b, h, currentDepth, -Infinity, Infinity, true, evalAiPlayer, evalHumanPlayer, deadlineMs, 0);

        if (result.aborted && bestMove !== -1) {
          break;
        }

        bestMove = result.move;
        bestScore = result.score;

        if (Math.abs(bestScore) >= 9000) break;
      }

      return { move: bestMove, score: bestScore };
    }

    // AI Move trigger
    function makeAiMove() {
      if (isGameOver || isAiThinking) return;
      isAiThinking = true;
      updateStatusDisplay();

      setTimeout(() => {
        const validMoves = [];
        for (let c = 0; c < 16; c++) {
          if (heights[c] < 4) validMoves.push(c);
        }

        if (validMoves.length === 0) {
          isAiThinking = false;
          return;
        }

        let chosenMove = -1;
        let moveScore = null;

        let targetDepth = 5;
        let timeLimitMs = 1500;

        if (difficulty === 'easy') {
          targetDepth = 3;
          timeLimitMs = 1000;
        } else if (difficulty === 'medium') {
          targetDepth = 5;
          timeLimitMs = 1500;
        } else if (difficulty === 'hard') {
          targetDepth = 7;
          timeLimitMs = 2000;
        } else if (difficulty === 'master') {
          const filled = heights.reduce((a, b) => a + b, 0);
          if (filled < 20) targetDepth = 8;
          else if (filled < 36) targetDepth = 10;
          else targetDepth = 12;
          timeLimitMs = 2500;
        } else if (difficulty === 'custom') {
          targetDepth = customDepth;
          timeLimitMs = Math.round(customTimeLimitSec * 1000);
        }

        if (difficulty === 'easy') {
          const candidateScores = validMoves.map(col => {
            const z = heights[col];
            const idx = col + z * 16;
            board[idx] = aiPlayer;
            heights[col]++;
            const score = evaluateBoard(board, heights, aiPlayer) + (Math.random() * 200 - 100);
            board[idx] = 0;
            heights[col]--;
            return { col, score };
          });
          candidateScores.sort((a, b) => b.score - a.score);
          chosenMove = candidateScores[0].col;
        } else {
          const searchRes = getBestMoveIterative(board, heights, targetDepth, timeLimitMs, aiPlayer, humanPlayer);
          chosenMove = searchRes.move;
          moveScore = searchRes.score;
        }

        if (chosenMove === -1 || heights[chosenMove] >= 4) {
          chosenMove = validMoves[Math.floor(Math.random() * validMoves.length)];
        }

        isAiThinking = false;

        if (chosenMove !== -1) {
          if (moveScore === null) {
            moveScore = evaluateBoard(board, heights, aiPlayer);
          }
          executeMove(chosenMove, moveScore);
        }
      }, 80);
    }

    // Calculate Coaching Scores for Human Player in Assist Mode (Strict 0 ~ 9999 Scale)
    function computeCoachingScores() {
      coachingScores.fill(null);
      if (!coachingMode || currentTurn !== humanPlayer || isGameOver) return;

      const validMoves = [];
      for (let c = 0; c < 16; c++) {
        if (heights[c] < 4) validMoves.push(c);
      }
      if (validMoves.length === 0) return;

      // Determine search depth for coaching assist
      let searchDepth = 5;
      if (difficulty === 'easy') searchDepth = 3;
      else if (difficulty === 'medium') searchDepth = 5;
      else if (difficulty === 'hard') searchDepth = 7;
      else if (difficulty === 'master') searchDepth = 8;
      else if (difficulty === 'custom') searchDepth = Math.min(customDepth, 9);

      // Check if AI already has immediate winning moves on current board
      const aiWinsNow = [];
      for (let col of validMoves) {
        const z = heights[col];
        const idx = col + z * 16;
        board[idx] = aiPlayer;
        if (checkWin(board)) {
          aiWinsNow.push(col);
        }
        board[idx] = 0;
      }

      for (let col of validMoves) {
        const z = heights[col];
        const idx = col + z * 16;

        // If AI has immediate win and human move doesn't block it -> Instant LOSE
        if (aiWinsNow.length > 0 && !aiWinsNow.includes(col)) {
          coachingScores[col] = -9999;
          continue;
        }
        if (aiWinsNow.length > 1) {
          coachingScores[col] = -9999; // Double threat -> unavoidable defeat
          continue;
        }

        board[idx] = humanPlayer;
        heights[col]++;

        const winState = checkWin(board);
        let score = 0;

        if (winState && winState.winner === humanPlayer) {
          score = 9999; // WIN
        } else {
          // Suicide Check: Does playing here allow AI to immediately win on top in the same column?
          let isSuicide = false;
          if (heights[col] < 4) {
            const topZ = heights[col];
            const topIdx = col + topZ * 16;
            board[topIdx] = aiPlayer;
            if (checkWin(board)) {
              isSuicide = true;
            }
            board[topIdx] = 0;
          }

          if (isSuicide) {
            score = -9999; // LOSE (Suicide move)
          } else {
            // Search from AI's perspective with iterative deepening and clean cache
            const aiRes = getBestMoveIterative(board, heights, searchDepth, 200, aiPlayer, humanPlayer);
            if (aiRes.score >= 8000) {
              score = -9999; // AI forced win sequence -> LOSE
            } else if (aiRes.score <= -8000) {
              score = 9999;  // Human forced win sequence -> WIN
            } else {
              score = -aiRes.score; // Human absolute perspective
            }
          }
        }

        board[idx] = 0;
        heights[col]--;

        // Clamp score strictly between -9999 and +9999
        if (score < -9000) score = -9999;
        else if (score > 9000) score = 9999;
        else score = Math.max(-8999, Math.min(8999, Math.round(score)));

        coachingScores[col] = score;
      }
    }

    // Execute Move (Human or AI)
    function executeMove(col, knownScore = null) {
      if (isGameOver || heights[col] >= 4 || isAiThinking) return;

      const level = heights[col];
      const cellIdx = col + level * 16;
      
      board[cellIdx] = currentTurn;
      heights[col]++;

      const win = checkWin(board);
      const isHuman = (currentTurn === humanPlayer);

      let score = knownScore;
      if (score === null) {
        score = evaluateBoard(board, heights, currentTurn);
      }

      history.push({
        turnNum: history.length + 1,
        isHuman: isHuman,
        colName: COL_NAMES[col],
        score: score,
        col: col,
        x: col % 4,
        y: Math.floor(col / 4),
        z: level
      });

      if (win) {
        isGameOver = true;
        winningLine = win.line;
      } else {
        const filled = heights.reduce((a, b) => a + b, 0);
        if (filled === 64) {
          isGameOver = true; // Draw
        } else {
          currentTurn = currentTurn === 1 ? 2 : 1;
        }
      }

      computeCoachingScores();
      renderUI();

      if (!isGameOver && currentTurn === aiPlayer) {
        makeAiMove();
      }
    }

    function resetGame() {
      board.fill(0);
      heights.fill(0);
      currentTurn = 1;
      isGameOver = false;
      winningLine = null;
      history = [];
      coachingScores.fill(null);
      isAiThinking = false;

      document.getElementById('btnHumanFirst').classList.toggle('active', humanPlayer === 1);
      document.getElementById('btnAIFirst').classList.toggle('active', humanPlayer === 2);

      computeCoachingScores();
      renderUI();

      if (currentTurn === aiPlayer) {
        makeAiMove();
      }
    }

    function autoAdjustCameraForDevice() {
      const w = window.innerWidth;
      const h = window.innerHeight;

      // Adjust camera distance for mobile / tablet screens so full 3D tower fits
      if (w <= 480) {
        cameraDistance = h > w ? 520 : 450;
      } else if (w <= 768) {
        cameraDistance = h > w ? 480 : 420;
      } else if (w <= 1100) {
        cameraDistance = 420;
      } else {
        cameraDistance = 380;
      }
    }

    // Touch Event Processing for Mobile & Tablet (Rotate, Pinch Zoom, Tap Move)
    let touchStartDist = 0;
    let touchStartX = 0;
    let touchStartY = 0;
    let isTouchTapCandidate = false;

    canvas.addEventListener('touchstart', (e) => {
      if (e.touches.length === 1) {
        isDragging = true;
        isTouchTapCandidate = true;
        lastMouseX = e.touches[0].clientX;
        lastMouseY = e.touches[0].clientY;
        touchStartX = lastMouseX;
        touchStartY = lastMouseY;
      } else if (e.touches.length === 2) {
        isDragging = false;
        isTouchTapCandidate = false;
        const dx = e.touches[0].clientX - e.touches[1].clientX;
        const dy = e.touches[0].clientY - e.touches[1].clientY;
        touchStartDist = Math.hypot(dx, dy);
      }
    }, { passive: false });

    canvas.addEventListener('touchmove', (e) => {
      e.preventDefault(); // Prevent body scroll while manipulating 3D stage

      if (e.touches.length === 1 && isDragging) {
        const cx = e.touches[0].clientX;
        const cy = e.touches[0].clientY;
        const dx = cx - lastMouseX;
        const dy = cy - lastMouseY;

        if (Math.hypot(cx - touchStartX, cy - touchStartY) > 10) {
          isTouchTapCandidate = false;
        }

        cameraAngleY += dx * 0.008;
        cameraAngleX += dy * 0.008;
        cameraAngleX = Math.max(-1.4, Math.min(1.4, cameraAngleX));

        lastMouseX = cx;
        lastMouseY = cy;
      } else if (e.touches.length === 2) {
        const dx = e.touches[0].clientX - e.touches[1].clientX;
        const dy = e.touches[0].clientY - e.touches[1].clientY;
        const newDist = Math.hypot(dx, dy);

        if (touchStartDist > 0) {
          const factor = touchStartDist / newDist;
          zoomCamera(factor > 1 ? 1.03 : 0.97);
          touchStartDist = newDist;
        }
      }
    }, { passive: false });

    canvas.addEventListener('touchend', (e) => {
      if (isTouchTapCandidate && currentTurn === humanPlayer) {
        const rect = canvas.getBoundingClientRect();
        const mx = touchStartX - rect.left;
        const my = touchStartY - rect.top;

        let closestCol = -1;
        let minDist = 35; // Touch tap tolerance

        for (let col = 0; col < 16; col++) {
          const cx = col % 4;
          const cy = Math.floor(col / 4);

          const pBottom = project(cx, cy, 0);
          const pTop = project(cx, cy, 3);

          const dist = distToSegment({ x: mx, y: my }, pBottom, pTop);
          if (dist < minDist) {
            minDist = dist;
            closestCol = col;
          }
        }

        if (closestCol !== -1 && heights[closestCol] < 4 && !isGameOver) {
          executeMove(closestCol);
        }
      }

      isDragging = false;
      isTouchTapCandidate = false;
      touchStartDist = 0;
    });


    // 3D Projection Math
    function project(x, y, z) {
      const gridSpacing = 52;
      // Center origin at 1.5, 1.5, 1.5
      const cx = (x - 1.5) * gridSpacing;
      const cy = (y - 1.5) * gridSpacing;
      const cz = (z - 1.5) * gridSpacing;

      // Yaw rotation (around Z-axis)
      const cosY = Math.cos(cameraAngleY);
      const sinY = Math.sin(cameraAngleY);
      const x1 = cx * cosY - cy * sinY;
      const y1 = cx * sinY + cy * cosY;
      const z1 = cz;

      // Pitch rotation (around rotated X-axis)
      const cosX = Math.cos(cameraAngleX);
      const sinX = Math.sin(cameraAngleX);
      const x2 = x1;
      const y2 = y1 * sinX - z1 * cosX;
      const z2 = y1 * cosX + z1 * sinX;

      // Perspective transformation
      const distance = cameraDistance;
      const scale = distance / (distance + y2);

      const w = canvas.width / (window.devicePixelRatio || 1);
      const h = canvas.height / (window.devicePixelRatio || 1);

      const px = w / 2 + x2 * scale;
      const py = h / 2 - z2 * scale;

      return { x: px, y: py, scale: scale, depth: y2 };
    }

    // Canvas Interaction (Rotate Camera)
    canvas.addEventListener('mousedown', (e) => {
      isDragging = true;
      lastMouseX = e.clientX;
      lastMouseY = e.clientY;
    });

    window.addEventListener('mouseup', () => {
      isDragging = false;

    window.addEventListener('mousemove', (e) => {

        const dx = e.clientX - lastMouseX;
        const dy = e.clientY - lastMouseY;

        cameraAngleY += dx * 0.008;
        cameraAngleX += dy * 0.008;

        // Clamp Pitch Angle
        cameraAngleX = Math.max(-1.4, Math.min(1.4, cameraAngleX));

        lastMouseX = e.clientX;
        lastMouseY = e.clientY;
      } else {
        // Raycast check for hover on 16 columns
        const rect = canvas.getBoundingClientRect();
        const mx = e.clientX - rect.left;
        const my = e.clientY - rect.top;

        let closestCol = -1;
        let minDist = 30;

        for (let col = 0; col < 16; col++) {
          const cx = col % 4;
          const cy = Math.floor(col / 4);

          const pBottom = project(cx, cy, 0);
          const pTop = project(cx, cy, 3);

          // Distance from point (mx, my) to pole line segment
          const dist = distToSegment({ x: mx, y: my }, pBottom, pTop);
          if (dist < minDist) {
            minDist = dist;
            closestCol = col;
          }
        }

        if (closestCol !== hoveredCol) {
          hoveredCol = closestCol;
          renderUI();
        }
      }

    function resetCamera() {
      autoAdjustCameraForDevice();
      cameraAngleX = 0.5;
      cameraAngleY = 0.78;
      cameraDistance = 380;
    }

    function zoomCamera(factor) {
      cameraDistance = Math.max(200, Math.min(700, cameraDistance * factor));
    }

    function distToSegment(p, v, w) {
      const l2 = (v.x - w.x) ** 2 + (v.y - w.y) ** 2;
      if (l2 === 0) return Math.hypot(p.x - v.x, p.y - v.y);
      let t = ((p.x - v.x) * (w.x - v.x) + (p.y - v.y) * (w.y - v.y)) / l2;
      t = Math.max(0, Math.min(1, t));
      return Math.hypot(p.x - (v.x + t * (w.x - v.x)), p.y - (v.y + t * (w.y - v.y)));
    }

    // Render 3D Scene Loop
    function draw3DScene() {
      const w = canvas.width / (window.devicePixelRatio || 1);
      const h = canvas.height / (window.devicePixelRatio || 1);

      ctx.clearRect(0, 0, w, h);

      const renderList = [];

      // 1. Base Grid Floor
      const p00 = project(-0.8, -0.8, -0.6);
      const p30 = project(3.8, -0.8, -0.6);
      const p33 = project(3.8, 3.8, -0.6);
      const p03 = project(-0.8, 3.8, -0.6);

      renderList.push({
        type: 'floor',
        depth: (p00.depth + p30.depth + p33.depth + p03.depth) / 4,
        draw: () => {
          ctx.beginPath();
          ctx.moveTo(p00.x, p00.y);
          ctx.lineTo(p30.x, p30.y);
          ctx.lineTo(p33.x, p33.y);
          ctx.lineTo(p03.x, p03.y);
          ctx.closePath();
          ctx.fillStyle = 'rgba(15, 23, 42, 0.7)';
          ctx.fill();
          ctx.strokeStyle = 'rgba(255, 255, 255, 0.15)';
          ctx.lineWidth = 2;
          ctx.stroke();

          // Floor grid lines
          for (let i = 0; i <= 3; i++) {
            const pa = project(i, 0, -0.55);
            const pb = project(i, 3, -0.55);
            ctx.beginPath();
            ctx.moveTo(pa.x, pa.y);
            ctx.lineTo(pb.x, pb.y);
            ctx.strokeStyle = 'rgba(255, 255, 255, 0.08)';
            ctx.lineWidth = 1;
            ctx.stroke();

            const pc = project(0, i, -0.55);
            const pd = project(3, i, -0.55);
            ctx.beginPath();
            ctx.moveTo(pc.x, pc.y);
            ctx.lineTo(pd.x, pd.y);
            ctx.stroke();
          }
        }
      });

      // 2. 16 Solid Poles
      for (let col = 0; col < 16; col++) {
        const cx = col % 4;
        const cy = Math.floor(col / 4);

        const pBot = project(cx, cy, -0.5);
        const pTop = project(cx, cy, 3.5);

        renderList.push({
          type: 'pole',
          col: col,
          depth: Math.max(pBot.depth, pTop.depth) + 0.001,
          draw: () => {
            const isHovered = hoveredCol === col;
            ctx.beginPath();
            ctx.moveTo(pBot.x, pBot.y);
            ctx.lineTo(pTop.x, pTop.y);
            ctx.lineWidth = Math.max(4, 7 * pBot.scale);
            ctx.strokeStyle = isHovered ? '#00f2ff' : '#475569';
            ctx.lineCap = 'round';
            ctx.stroke();

            // Pole cap
            ctx.beginPath();
            ctx.arc(pTop.x, pTop.y, 4 * pTop.scale, 0, Math.PI * 2);
            ctx.fillStyle = isHovered ? '#00f2ff' : '#94a3b8';
            ctx.fill();
          }
        });
      }

      // 3. Placed Spheres
      const t = COLOR_THEMES[currentThemeKey];

      for (let idx = 0; idx < 64; idx++) {
        const val = board[idx];
        if (val === 0) continue;

        const cx = idx % 4;
        const cy = Math.floor(idx / 4) % 4;
        const cz = Math.floor(idx / 16);

        const p = project(cx, cy, cz);
        const radius = 18 * p.scale;

        renderList.push({
          type: 'sphere',
          depth: p.depth,
          draw: () => {
            const colors = val === 1 ? t.p1 : t.p2;

            // 3D Radial Gradient Shading
            const grad = ctx.createRadialGradient(
              p.x - radius * 0.3,
              p.y - radius * 0.3,
              radius * 0.1,
              p.x,
              p.y,
              radius
            );
            grad.addColorStop(0, colors.light);
            grad.addColorStop(0.7, colors.base);
            grad.addColorStop(1, colors.dark);

            ctx.beginPath();
            ctx.arc(p.x, p.y, radius, 0, Math.PI * 2);
            ctx.fillStyle = grad;
            ctx.fill();

            ctx.strokeStyle = 'rgba(0, 0, 0, 0.3)';
            ctx.lineWidth = 1;
            ctx.stroke();
          }
        });
      }

      // 4. Hover Preview Sphere
      if (hoveredCol !== -1 && heights[hoveredCol] < 4 && !isGameOver && currentTurn === humanPlayer) {
        const cx = hoveredCol % 4;
        const cy = Math.floor(hoveredCol / 4);
        const cz = heights[hoveredCol];

        const p = project(cx, cy, cz);
        const radius = 18 * p.scale;

        renderList.push({
          type: 'preview',
          depth: p.depth - 1, // Draw slightly in front
          draw: () => {
            const colors = currentTurn === 1 ? t.p1 : t.p2;
            ctx.save();
            ctx.globalAlpha = 0.5;

            ctx.beginPath();
            ctx.arc(p.x, p.y, radius, 0, Math.PI * 2);
            ctx.fillStyle = colors.light;
            ctx.fill();

            ctx.strokeStyle = '#fff';
            ctx.lineWidth = 2;
            ctx.stroke();
            ctx.restore();
          }
        });
      }

      // 5. Teaching Mode Scores on 3D Canvas
      if (coachingMode && currentTurn === humanPlayer && !isGameOver) {
        for (let col = 0; col < 16; col++) {
          if (heights[col] >= 4 || coachingScores[col] === null) continue;

          const cx = col % 4;
          const cy = Math.floor(col / 4);

          const p = project(cx, cy, 3.8);
          const score = coachingScores[col];

          renderList.push({
            type: 'coachingLabel',
            depth: p.depth - 2,
            draw: () => {
              let scoreStr = score > 0 ? `+${score}` : `${score}`;
              if (score >= 9000) scoreStr = 'WIN';
              else if (score <= -9000) scoreStr = 'LOSE';

              let bg = 'rgba(245, 158, 11, 0.85)';
              if (score >= 30) bg = 'rgba(16, 185, 129, 0.85)';
              else if (score <= -150) bg = 'rgba(239, 68, 68, 0.85)';

              ctx.font = 'bold 11px sans-serif';
              const textWidth = ctx.measureText(scoreStr).width;

              ctx.fillStyle = bg;
              ctx.beginPath();
              ctx.roundRect(p.x - textWidth / 2 - 5, p.y - 10, textWidth + 10, 18, 6);
              ctx.fill();

              ctx.fillStyle = '#ffffff';
              ctx.textAlign = 'center';
              ctx.textBaseline = 'middle';
              ctx.fillText(scoreStr, p.x, p.y);
            }
          });
        }
      }

      // 6. Winning Line Glow Highlight
      if (winningLine) {
        const idx0 = winningLine[0];
        const idx3 = winningLine[3];

        const p0 = project(idx0 % 4, Math.floor(idx0 / 4) % 4, Math.floor(idx0 / 16));
        const p3 = project(idx3 % 4, Math.floor(idx3 / 4) % 4, Math.floor(idx3 / 16));

        renderList.push({
          type: 'winningLine',
          depth: -9999, // Render on top
          draw: () => {
            ctx.save();
            ctx.beginPath();
            ctx.moveTo(p0.x, p0.y);
            ctx.lineTo(p3.x, p3.y);
            ctx.lineWidth = 10;
            ctx.strokeStyle = 'rgba(52, 211, 153, 0.8)';
            ctx.lineCap = 'round';
            ctx.shadowColor = '#34d399';
            ctx.shadowBlur = 15;
            ctx.stroke();
            ctx.restore();
          }
        });
      }

      // Painter's Algorithm: Sort by depth back-to-front
      renderList.sort((a, b) => b.depth - a.depth);

      renderList.forEach(obj => obj.draw());

      requestAnimationFrame(draw3DScene);
    }

    // Check environment: Show download button ONLY on Developer URL / Local
    const isDevEnv = window.location.hostname.includes('ais-dev') || 
                     window.location.hostname === 'localhost' || 
                     window.location.hostname === '127.0.0.1';
    const docBtn = document.getElementById('docDownloadBtn');
    if (docBtn) {
      docBtn.style.display = isDevEnv ? 'inline-flex' : 'none';
    }

    // Start 3D Loop
    requestAnimationFrame(draw3DScene);

    // Initial Setup
    resetGame();
  </script>
